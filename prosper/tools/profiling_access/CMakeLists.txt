cmake_minimum_required(VERSION 3.20)
project(prosper_profiling_access_checks LANGUAGES NONE)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The profiling access helper and its tests are Linux-only")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME profiling_access COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_access.py")
set_tests_properties(profiling_access PROPERTIES TIMEOUT 30)
